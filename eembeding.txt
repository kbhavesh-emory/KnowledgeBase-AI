🚀 Examples
# 1) Sync using settings.GITHUB_REPOS from backend/config/settings.py
python -m backend.scripts.sync_repositories

# 2) Sync explicit repos (shallow clone)
python -m backend.scripts.sync_repositories \
  --repo https://github.com/pallets/flask \
  --repo https://github.com/tiangolo/fastapi \
  --shallow

# 3) Sync from file, then embed
python -m backend.scripts.sync_repositories --repos-file repos.txt
python -m backend.scripts.embed_repositories --repos data/repositories --glob "**/*"

# 4) One-shot: Sync + Embed
python -m backend.scripts.sync_and_embed \
  --repo https://github.com/pallets/flask \
  --repo https://github.com/tiangolo/fastapi \
  --shallow --glob "**/*.{py,md,txt,json,yaml}"
